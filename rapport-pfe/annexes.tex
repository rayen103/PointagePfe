\section{Database Table Creation Scripts (DDL)}
Below is the SQL DDL script defining the core tables in the SQL Server database schema, illustrating indices and multi-tenant keys:

\begin{lstlisting}[language=SQL, caption={SQL Server DDL Database Schema definition}]
-- Create the main schema
CREATE SCHEMA F3SManagement;
GO

-- Create the Societes Table
CREATE TABLE F3SManagement.Societes (
    SocieteId VARCHAR(26) NOT NULL PRIMARY KEY,
    Nom VARCHAR(100) NOT NULL,
    CodeSociete VARCHAR(20) NOT NULL UNIQUE,
    MatriculeFiscal VARCHAR(50) NULL,
    LogoPath VARCHAR(255) NULL,
    InsererPar VARCHAR(50) NULL,
    DateInsertion DATETIME NULL,
    ModifierPar VARCHAR(50) NULL,
    DateModification DATETIME NULL
);

-- Create the Buses Table
CREATE TABLE F3SManagement.Buses (
    BusId VARCHAR(26) NOT NULL PRIMARY KEY,
    NumeroIMM VARCHAR(50) NOT NULL,
    ModelBus VARCHAR(100) NULL,
    IMEI VARCHAR(15) NULL,
    Capacite INT NULL,
    CodeCircuit VARCHAR(50) NULL,
    CodeChauffeur VARCHAR(50) NULL,
    IsActive BIT NOT NULL DEFAULT 1,
    Latitude DECIMAL(9,6) NULL,
    Longitude DECIMAL(9,6) NULL,
    CurrentOccupancy INT NOT NULL DEFAULT 0,
    LastPositionAt DATETIME NULL,
    LastOccupancyUpdateAt DATETIME NULL,
    SocieteId VARCHAR(26) NOT NULL,
    CONSTRAINT FK_Buses_Societes FOREIGN KEY (SocieteId) 
        REFERENCES F3SManagement.Societes(SocieteId)
);

-- Create the Employes Table
CREATE TABLE F3SManagement.Employes (
    EmployeId VARCHAR(26) NOT NULL PRIMARY KEY,
    Matricule VARCHAR(20) NOT NULL,
    RFID VARCHAR(50) NULL,
    Nom VARCHAR(50) NOT NULL,
    Prenom VARCHAR(50) NOT NULL,
    CodeCircuit VARCHAR(20) NULL,
    CodePointCollecte VARCHAR(20) NULL,
    CodeShift VARCHAR(20) NULL,
    SocieteId VARCHAR(26) NOT NULL,
    CONSTRAINT FK_Employes_Societes FOREIGN KEY (SocieteId) 
        REFERENCES F3SManagement.Societes(SocieteId)
);

-- Create the BusRuntimeEvents Table
CREATE TABLE F3SManagement.BusRuntimeEvents (
    BusRuntimeEventId VARCHAR(26) NOT NULL PRIMARY KEY,
    BusId VARCHAR(26) NOT NULL,
    EventType VARCHAR(50) NOT NULL,
    Description VARCHAR(255) NOT NULL,
    Latitude DECIMAL(9,6) NULL,
    Longitude DECIMAL(9,6) NULL,
    Occupancy INT NOT NULL,
    OccurredAtUtc DATETIME NOT NULL,
    CONSTRAINT FK_BusRuntimeEvents_Buses FOREIGN KEY (BusId) 
        REFERENCES F3SManagement.Buses(BusId)
);

-- Create indices for multi-tenant query optimizations
CREATE INDEX IX_Buses_SocieteId ON F3SManagement.Buses(SocieteId);
CREATE INDEX IX_Employes_SocieteId ON F3SManagement.Employes(SocieteId);
CREATE INDEX IX_BusRuntimeEvents_BusId ON F3SManagement.BusRuntimeEvents(BusId);
\end{lstlisting}
